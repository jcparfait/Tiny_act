  has_many :user_interests, dependent: :destroy
  has_many :interests, through: :user_interests

  devise :database_authenticatable, :registerable,
         :recoverable, :rememberable, :validatable

  after_create :create_default_room

  private

  def create_default_room
    create_room!(width: Room::GRID_WIDTH, height: Room::GRID_HEIGHT)
  end
end

class ApplicationController < ActionController::Base
  before_action :authenticate_user!

  # Avant chaque action d'un controller Devise (inscription/connexion),
  # on exécute la méthode ci-dessous.
  before_action :configure_permitted_parameters, if: :devise_controller?

  def default_url_options
    { host: ENV["DOMAIN"] || "localhost:3000" }
  end

  private

  # On autorise first_name et last_name à être enregistrés
  # UNIQUEMENT au moment de l'inscription (:sign_up).
  def configure_permitted_parameters
    devise_parameter_sanitizer.permit(:sign_up, keys: %i[first_name last_name])
  end

    class ActivitySessionProgressController < ApplicationController
      skip_before_action :authenticate_user!
      skip_forgery_protection

      before_action :set_activity_session

      def show
        render json: {
          activity_session_id: @activity_session.id,
          progress_data: @activity_session.progress_data || {}
        }
      end

      def update
        merged_progress = current_progress.deep_merge(progress_params.to_h)

        @activity_session.update!(
          progress_data: merged_progress
        )

        render json: {
          activity_session_id: @activity_session.id,
          progress_data: @activity_session.progress_data
        }
      end

      private

      def set_activity_session
        user = current_api_user

        unless user
          render json: {
            error: "Aucun utilisateur disponible pour le test API."
          }, status: :unprocessable_entity

          return
        end

        @activity_session = user.activity_sessions.find(
          params[:activity_session_id]
        )
      end

      def current_api_user
        User
          .joins(:user_interests)
          .distinct
          .first || User.first
      end

      def current_progress
        @activity_session.progress_data.presence || {}
      end

      def progress_params
        params.require(:progress_data).permit!
      end
    end
  end

const API_URL = process.env.EXPO_PUBLIC_API_URL;

async function fetchJson<T>(url: string, options?: RequestInit): Promise<T> {
  const response = await fetch(url, options);
  const text = await response.text();

  let data: unknown = null;

  if (text.length > 0) {
    try {
      data = JSON.parse(text);
    } catch {
      const preview = text.slice(0, 160).replace(/\s+/g, " ");

      throw new Error(
        `Rails n'a pas renvoyé du JSON. Status ${response.status}. URL appelée : ${url}. Début de réponse : ${preview}`
      );
    }
  }

  if (!response.ok) {
    const errorMessage =
      data &&
      typeof data === "object" &&
      "error" in data &&
      typeof data.error === "string"
        ? data.error
        : `Erreur API : ${response.status}`;

    throw new Error(errorMessage);
  }

  return data as T;
}

export async function loadInitialData(): Promise<InitialDataResponse> {
  const [moods, locations, durations] = await Promise.all([
    fetchJson<Mood[]>(`${API_URL}/api/v1/moods`),
    fetchJson<Location[]>(`${API_URL}/api/v1/locations`),
    fetchJson<Duration[]>(`${API_URL}/api/v1/durations`),
  ]);

  return {
    moods,
    locations,
    durations,
  };
}
